import type { GameState } from '../../game/types'

const BASE_TO_VARIANTS: Record<string, string[]> = {
  hollow: ['hollow-4'],
  lessons: ['lessons-4'],
  // Add other 4p variants as cards in F9 require them.
  // Cards consuming this helper: C130 (grove/hollow), B152 (lessons/lessons-4),
  // C39 (traveling-players, no variant), C15 (pig-market, no variant),
  // E166 (no specific 4p variant per BGA spec).
}

/**
 * Resolve a base action space id to the actual list of space ids in this game,
 * including 4p extension variants (hollow-4, lessons-4) when they exist.
 *
 * Used by listeners that match on action space id for cards whose BGA logic
 * keys on `actionCardType` (which is invariant across 1-3p / 4p variants),
 * but our side keys on the literal space id string.
 */
export const pairedSpaceIdFor = (
  state: GameState,
  baseId: string,
): string[] => {
  const variants = BASE_TO_VARIANTS[baseId] ?? []
  const presentVariants = variants.filter((v) =>
    (state.actionSpaces ?? []).some((s) => s.id === v),
  )
  return [baseId, ...presentVariants]
}
```

import type { ActionDefinition, ActionExecutionResult } from '../../../game/types'

export const takeFromSpaceAction: ActionDefinition = {
  id: 'take-from-space',
  nameKey: 'actions.take-from-space',
  flow: { type: 'leaf', actionId: 'take-from-space' },
  canBeExecutedByPlayer: () => true,
  execute: (context): ActionExecutionResult => {
    const { state, player, actionContext } = context as {
      state: { actionSpaces: { id: string; resources?: Record<string, number> }[] }
      player: { resources: Record<string, number> }
      actionContext?: { spaceId?: string; resource?: string; amount?: number }
    }
    const spaceId = actionContext?.spaceId
    const resource = actionContext?.resource
    const amount = actionContext?.amount ?? 1
    if (!spaceId || !resource || amount <= 0) {
      return { type: 'fail', logKey: 'log.takeFromSpaceInvalid' }
    }
    const space = state.actionSpaces.find((s) => s.id === spaceId)
    if (!space?.resources) {
      return { type: 'fail', logKey: 'log.takeFromSpaceNoSpace' }
    }
    const have = space.resources[resource] ?? 0
    if (have < amount) {
      return { type: 'fail', logKey: 'log.takeFromSpaceNotEnough' }
    }
    space.resources[resource] = have - amount
    player.resources[resource] = (player.resources[resource] ?? 0) + amount
    return { type: 'ok', resourcesGained: { [resource]: amount } as Record<string, number> }
  },
}
```
