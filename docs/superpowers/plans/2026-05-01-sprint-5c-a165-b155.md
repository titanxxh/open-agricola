export const applyTradeSideEffect = (
  state: GameState,
  eff: TradeSideEffect,
  times: number,

  if (times <= 0) return
  switch (eff.type) {
    case 'drainSpace': {
      const space = state.actionSpaces.find((s) => s.id === eff.spaceId)
      if (!space?.resources) return

      space.resources[eff.resource] = Math.max(0, cur - times)
      return
    }
  }
}

```ts
// shared/actions/effects/breed.ts
import type {
  ActionDefinition,
  ActionExecutionResult,
  ActionFlow,
  GameState,
  HarvestBreedSummary,
  PlayerState,
} from '../../game/types'
import { getAssignedAnimalCount, getTotalAnimalCapacity } from '../helpers/animal-zones'

export type BreedAnimalType = 'sheep' | 'boar' | 'cattle'

export type BreedOptions = {
  animalTypes?: ReadonlyArray<BreedAnimalType>
  sourceCard: string
}

const DEFAULT_TYPES: ReadonlyArray<BreedAnimalType> = ['sheep', 'boar', 'cattle']

export const breed = (
  state: GameState,
  player: PlayerState,
  opts: BreedOptions,
): { breedSummary: HarvestBreedSummary } => {
  const types = opts.animalTypes ?? DEFAULT_TYPES
  const summary: HarvestBreedSummary = { resources: {}, animalTypes: 0, animalCount: 0 }
  for (const type of types) {
    const free = getTotalAnimalCapacity(player) - getAssignedAnimalCount(player)
    if (free <= 0) break
    if (player.resources[type] < 2) continue
    player.resources[type] += 1
    summary.resources[type] = 1
    summary.animalTypes += 1
    summary.animalCount += 1
  }
  return { breedSummary: summary }
}

export const breedAction: ActionDefinition = {
  id: 'breed',
  nameKey: 'actions.breed.name',
  descriptionKey: 'actions.breed.description',
  roundAvailable: 1,
  gainPerRound: {},
  canBeExecutedByPlayer: () => true,
  execute: ({ state, player, actionContext }): ActionExecutionResult => {
    const ctx = (actionContext ?? {}) as { animalTypes?: BreedAnimalType[]; sourceCard?: string }
    const sourceCard = ctx.sourceCard ?? 'unknown'
    const { breedSummary } = breed(state, player, {
      animalTypes: ctx.animalTypes,
      sourceCard,
    })
    if (sourceCard === 'harvest') {
      state.harvestBreedSummary ??= {}
      state.harvestBreedSummary[player.id] = breedSummary
    }
    if (breedSummary.animalCount > 0) {
      return { type: 'animalReorg', sourceId: `card:${sourceCard}` }
    }
    return { type: 'ok' }
  },
}

export const breedLeaf = (
  sourceCard: string,
  animalTypes?: ReadonlyArray<BreedAnimalType>,
): ActionFlow => ({
  type: 'leaf',
  actionId: 'breed',
  actionContext: { animalTypes: animalTypes ?? null, sourceCard },
  sourceCard,
})
```

    children: players.flatMap((p) => [
      { type: 'playerSwitch' as const, targetPlayerId: p.id },
      breedLeaf('harvest'),

  }
