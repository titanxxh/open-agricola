# Sprint 7b1 — Pay-as-effect Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `pay` a first-class engine effect; migrate 8 BGA `onPlayerAfterPay`-style cards (E123/E54/E122/E128/C116/B18/D74/C148) to `actions:['pay']`; reclaim C146 from 7b deferred via existing C104 multi-select pattern.

**Architecture:** Replace `pay-resources` with new generic `pay` ActionDefinition that handles `Partial<Resource>` + ComplexCost + paymentSolution branches. Rewrite `improvement / renovation / occupation` action flows as `seq:[pay, apply-*]`. Convert `construct + farm-choice` to engine-driven (deprecate `actionContext.farmPayment`). Replace `_activeActionBonusSources` with `extraData.bonusUsed` transparency. Add `PaymentSolution.bonusChoiceIndex` for E123 use-top-k.

**Tech Stack:** TypeScript 5.x / Vitest 4 / pnpm 10 / Vite 7 / Node 22. No new dependencies.

```ts
import { resolveCostPaymentSelection, executeResolvedTypedFlatPayment } from '../helpers/pay-helpers'

execute: ({ player, params }) => {
  const p = params as PayParams | undefined
  if (!p?.cost) return { type: 'fail', logKey: 'log.payFail' }
  if (isComplexCost(p.cost)) {
    const optionPrefix = p.optionPrefix ?? 'pay:generic'
    const selection = resolveCostPaymentSelection(player, p.cost, optionPrefix, p.paymentChoice, { type: 'fail', logKey: 'log.payFail' }, { costType: p.costType, includeReturnedCard: p.includeReturnedCard })
    if ('type' in selection && selection.type !== 'selected') {
      // selection 是 ActionExecutionResult 'choice' 或 'fail'
      return selection
    }
    if (selection.type === 'selected') {
      executePaymentSolution(player, selection.solution, { costType: p.costType })
      return { type: 'ok', extraData: { bonusUsed: selection.solution.bonusUsed?.split(',') ?? [], bonusChoiceIndex: selection.solution.bonusChoiceIndex } }
    }
  }
  // simple Partial<Resource> 分支已在 Task 1.4 实现
  ...
},
resolveChoice: ({ player, params }, choice) => {
  const p = params as PayParams | undefined
  if (!p?.cost) return { type: 'fail' }
  // 重入 execute 把 paymentChoice 设进去
  return payAction.execute({ player, params: { ...p, paymentChoice: choice }, ... } as any)
},
```

```ts
const PAIRS: readonly ['WC' | 'WR' | 'WS' | 'CR' | 'CS' | 'RS', Partial<Resource>][] = [
  ['WC', { wood: 1, clay: 1 }],
  ['WR', { wood: 1, reed: 1 }],
  ['WS', { wood: 1, stone: 1 }],
  ['CR', { clay: 1, reed: 1 }],
  ['CS', { clay: 1, stone: 1 }],
  ['RS', { reed: 1, stone: 1 }],
]

const PAIR_RESOURCES: Record<string, Partial<Resource>> = Object.fromEntries(PAIRS)

const buildAllPairsGain = (): Partial<Resource> => {
  const result: Partial<Resource> = {}
  for (const [, res] of PAIRS) {
    for (const [k, v] of Object.entries(res)) {
      result[k as keyof Resource] = (result[k as keyof Resource] ?? 0) + (v ?? 0)
    }
  }
  return result
}

const onBuy: CardEffect['onBuy'] = (state, player) => {
  const n = Math.min(6, countAllImprovements(player))
  if (n === 0) return
  if (n === 6) {
    return gainLeaf(CARD_ID, buildAllPairsGain())
  }
  return {
    type: 'choice' as const,
    promptKey: 'ui.interactionWorkshopAssistantSelect',
    promptParams: { needed: n },
    options: PAIRS.map(([k]) => ({
      value: k,
      labelKey: `cards.C146.pair.${k}`,
    })),
  }
}

const onBuyResolveChoice: CardEffect['onBuyResolveChoice'] = (state, player, choice) => {
  const selected = [...new Set(choice.split(','))].filter((k) => k in PAIR_RESOURCES)
  const n = Math.min(6, countAllImprovements(player))
  if (selected.length !== n) {
    // re-emit
    return {
      type: 'choice' as const,
      promptKey: 'ui.interactionWorkshopAssistantSelect',
      promptParams: { needed: n },
      options: PAIRS.map(([k]) => ({ value: k, labelKey: `cards.C146.pair.${k}` })),
    }
  }
  const gains: Partial<Resource> = {}
  for (const k of selected) {
    for (const [r, amt] of Object.entries(PAIR_RESOURCES[k]!)) {
      gains[r as keyof Resource] = (gains[r as keyof Resource] ?? 0) + (amt ?? 0)
    }
  }
  return gainLeaf(CARD_ID, gains)
}
```
