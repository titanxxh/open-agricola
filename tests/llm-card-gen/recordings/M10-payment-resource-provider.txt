```typescript
const CARD_ID = 'CUSTOM_TravelingTeacher'

const CARD_DEF = {
  cardType: 'occupation',
  meta: {
    id: CARD_ID,
    name: 'Traveling Teacher',
    deck: 'CUSTOM',
    number: 0,
    desc: ['When you pay costs for an Occupation, you may use food from the Traveling Players space to pay. Remove the used food from that space.'],
    cost: {},
    vp: 0,
    implemented: true,
    locales: {
      zh: {
        name: '巡游教师',
        desc: ['当你支付职业费用时，可以使用 Traveling Players 行动格上的食物来支付。从该行动格移除所使用的食物。'],
      },
    },
  },
}

const CARD_IMPL = {
  listeners: [
    {
      cardIds: [CARD_ID],
      actions: ['occupation'],
      phases: ['computeCosts'],
      handler: (context) => {
        const travelingPlayersSpace = context.state.actionSpaces.find(s => s.id === 'traveling-players')
        if (!travelingPlayersSpace) return
        const availableFood = travelingPlayersSpace.resources?.food ?? 0
        if (availableFood <= 0) return
        return {
          paymentResourceProviders: [{
            key: `${CARD_ID}:traveling-players-food`,
            sourceCard: CARD_ID,
            available: availableFood,
            covers: [{ resource: 'food', costAmount: 1, paymentAmount: 1 }],
            consume: { type: 'actionSpace', spaceId: 'traveling-players', resource: 'food' },
          }],
          sourceCard: CARD_ID,
        }
      },
    },
  ],
}
```
